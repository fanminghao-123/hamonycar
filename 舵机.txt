#include <stdio.h>
#include <unistd.h>
#include "ohos_init.h"
#include "cmsis_os2.h"
#include "wifiiot_gpio.h"
#include "wifiiot_gpio_ex.h"
#include "hi_io.h"
#include "hi_time.h"

osMutexId_t mutex_id;
uint8_t flag = 0;
#define GPIO2 2

void set_angle( unsigned int duty) {
    GpioSetOutputVal(GPIO2, WIFI_IOT_GPIO_VALUE1);
    hi_udelay(duty);
    GpioSetOutputVal(GPIO2, WIFI_IOT_GPIO_VALUE0);
    hi_udelay(20000 - duty);
}

void engine_run_0(void)
{
    for (int i = 0; i <10; i++)
    {
        set_angle(500);
    }
}

void engine_run_90(void)
{
    for (int i = 0; i <10; i++)
    {
        set_angle(1500);
    }
}

void engine_run_180(void)
{
    for (int i = 0; i <10; i++)
    {
        set_angle(2500);
    }
}

/* thread1：转到180度 */
static void __attribute__((unused)) thread1(void)
{
    osDelay(100U);
    while (1)
    {
        osMutexAcquire(mutex_id, osWaitForever);
        printf("thread1 run: turn to 180°\r\n");
        flag = 180;
        engine_run_180();
        osDelay(100U);   //180°停留1秒
        osMutexRelease(mutex_id);

        osDelay(50U);  //释放锁休眠0.5秒，让出锁给thread3
    }
}

/* thread2：打印舵机状态，不控制舵机 */
static void __attribute__((unused)) thread2(void)
{
    osDelay(100U);
    while (1)
    {
        osMutexAcquire(mutex_id, osWaitForever);
        switch(flag)
        {
        case 90 :
            printf("SG90 stay at 90 du.\r\n");break;
        case 180 :
            printf("SG90 stay at 180 du.\r\n");break;
        case 0 :
            printf("SG90 stay at 0 du.\r\n");break;
        default : break;
        }
        flag = 0;
        osMutexRelease(mutex_id);
        osDelay(50U);
    }
}

/* thread3：转到0度 */
static void __attribute__((unused)) thread3(void)
{
    osDelay(200U);
    while (1)
    {
        osMutexAcquire(mutex_id, osWaitForever);
        printf("thread3 run: turn to 0°\r\n");
        flag = 0;
        engine_run_0();
        osDelay(100U);  //0°停留1秒
        osMutexRelease(mutex_id);

        osDelay(50U); //释放锁休眠0.5秒
    }
}

static void SG90(void)
{
    GpioInit();
    IoSetFunc(WIFI_IOT_IO_NAME_GPIO_2,WIFI_IOT_IO_FUNC_GPIO_2_GPIO);
    GpioSetDir(WIFI_IOT_IO_NAME_GPIO_2,WIFI_IOT_GPIO_DIR_OUT);

    mutex_id = osMutexNew(NULL);
    if (mutex_id == NULL)
    {
        printf("Failed to create Mutex!\n");
        return;
    }

    osThreadAttr_t attr;
    attr.attr_bits = 0U;
    attr.cb_mem = NULL;
    attr.cb_size = 0U;
    attr.stack_mem = NULL;
    attr.stack_size = 1024 * 4;

    attr.name = "thread1";
    attr.priority = 26;
    if (osThreadNew((osThreadFunc_t)thread1, NULL, &attr) == NULL)
    {
        printf("Failed to create thread1!\n");
    }

    attr.name = "thread2";
    attr.priority = 25;
    if (osThreadNew((osThreadFunc_t)thread2, NULL, &attr) == NULL)
    {
        printf("Failed to create thread2!\n");
    }

    attr.name = "thread3";
    attr.priority = 24;
    if (osThreadNew((osThreadFunc_t)thread3, NULL, &attr) == NULL)
    {
        printf("Failed to create thread3!\n");
    }
}

APP_FEATURE_INIT(SG90);
